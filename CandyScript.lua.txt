-- =================================================

local activeTweens = {}

local function playTween(obj, info, props)
    if activeTweens[obj] then
        activeTweens[obj]:Cancel()
    end
    
    local tween = TweenService:Create(obj, info, props)
    activeTweens[obj] = tween
    tween:Play()
    
    tween.Completed:Connect(function()
        if activeTweens[obj] == tween then
            activeTweens[obj] = nil
        end
    end)
end

local dragging = false
local dragInput, dragStart, startPos

main.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

main.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

RunService.RenderStepped:Connect(function()
    if dragging and dragInput then
        local delta = dragInput.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

local candies = {"Candy1","Candy2","Candy3"}

local function getCandyRoot(name)
    local obj = workspace:FindFirstChild(name, true)
    if not obj then return end
    return obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
end

local currentMode = "Directo"
local currentColor = Color3.fromRGB(255,100,180)

local function fly(mode, candy)
    local char = player.Character or player.CharacterAdded:Wait()
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local root = getCandyRoot(candy)
    if not root then return end
    
    local target = root.Position + Vector3.new(0,5,0)
    
    if mode == "Directo" then
        
        playTween(hrp,
            TweenInfo.new(1.5, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
            {CFrame = CFrame.new(target)}
        )
        
    else -- SUBIR MODO ÉPICO
        
        local startPos = hrp.Position
        
        -- 1 Subir 100 studs
        local upPos = startPos + Vector3.new(0,100,0)
        
        playTween(hrp,
            TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {CFrame = CFrame.new(upPos)}
        )
        
        task.wait(0.9)
        
        -- 2 Moverse arriba del objetivo
        local aboveTarget = Vector3.new(target.X, target.Y + 100, target.Z)
        
        playTween(hrp,
            TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {CFrame = CFrame.new(aboveTarget)}
        )
        
        task.wait(0.9)
        
        -- 3 Bajada rápida
        playTween(hrp,
            TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            {CFrame = CFrame.new(target)}
        )
        
        task.wait(0.4)
        
        -- 4 Esperar 2 segundos abajo
        task.wait(2)
        
        -- 5 Volver a subir
        playTween(hrp,
            TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {CFrame = CFrame.new(target + Vector3.new(0,100,0))}
        )
    end
end

local connections = {}

local function limpiar()
    for _,c in pairs(connections) do
        c:Disconnect()
    end
    table.clear(connections)
    content:ClearAllChildren()
end

local function crearBotones()
    limpiar()
    
    for _, candy in ipairs(candies) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.85,0,0,70)
        btn.Position = UDim2.new(0.075,0,0,0)
        btn.BackgroundColor3 = Color3.fromRGB(255,235,245)
        btn.Text = currentMode.." → "..candy
        btn.TextColor3 = currentColor
        btn.Font = Enum.Font.GothamSemibold
        btn.TextSize = 18
        btn.AutoButtonColor = false
        btn.Parent = content
        
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0,20)
        
        local stroke = Instance.new("UIStroke", btn)
        stroke.Color = currentColor
        stroke.Thickness = 2
        stroke.Transparency = 0.7
        
        table.insert(connections,
            btn.MouseButton1Click:Connect(function()
                fly(currentMode, candy)
            end)
        )
    end
end

local function crearSidebarBtn(text,y,color)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1,-20,0,70)
    btn.Position = UDim2.new(0,10,0,y)
    btn.BackgroundColor3 = Color3.fromRGB(255,230,240)

    btn.MouseButton1Click:Connect(function()
        currentMode = text
        currentColor = color

    end)
end
